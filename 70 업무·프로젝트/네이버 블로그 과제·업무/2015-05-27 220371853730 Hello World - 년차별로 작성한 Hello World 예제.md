******5년차 :**

![](https://postfiles.pstatic.net/20150527_7/agapeuni_1432712349032avIS8_PNG/5.png?type=w80_blur)

public class HelloWorld { public static final String PHRASE_HELLO = "Hello"; public static final String PHRASE_WORLD = "World!"; public String hello = null; public String world = null; public HelloWorld(String hello, String world) { this.hello = hello; this.world = world; } public void sayPhrase() { String first = this.hello.toString(); String second = this.world.toString(); System.out.println(first + " " + second); } public void setHello(String h) { this.hello = h } public String getHello() { return this.hello; } public void setWorld(String w) { this.world = w; } public String getWorld() { return this.world; } public static void main(String[] args) { DIC d = DependencyInjectionContainer::factory(); HelloWorld hw = null; try { hw = d.newInstance(HelloWorld.class); } catch (Exception e) { e.printStackTrace(); } try { hw.setHello(PHRASE_HELLO); hw.setWorld(PHRASE_HELLO); hw.sayPhrase(); } catch (Exception e) { e.printStackTrace(); } } }
