1\. MEAN 스택으로 칸반(Kanban) 구현

4\. Kanban 구현

MongoDB를 사용하기 위해 mongoose 모듈을 사용하였다.

파일 : ds.js

var mongoose = require('mongoose'); exports.connect = function() { mongoose.connect('mongodb://localhost/kanban'); mongoose.connection.on('connected', function() { console.log('mongoose Connect OK'); }); mongoose.connection.on('error', function(err) { console.log('mongoose Connect Fail, err = ' + err); }); mongoose.connection.on('disconnected', function() { console.log('disconnected'); }); process.on('SIGINT', function() { mongoose.connection.close(function() { console.log('connection.close...'); process.exit(0); }); }); };

라우팅 정보와 task-constroller.js의 메서드와 연결한다.

파일 : router.js

var task = require('../controllers/task-controller.js'); exports.route = function(app) { app.get('/', task.list); app.get('/listTask', task.list); app.post('/createTask', task.create); app.post('/updateTask', task.update); app.post('/removeTask', task.remove); };

​

라우팅 정보와 연계하여 실제로 동작하는 코드가 있다. task-model.js 참조하여 데이터 처리를 수행한다.

칸반의 3가지 단계(To-Do, Doing, Done)에서 스크럼의 백로그와 유사한 개념으로 PLAN 단계를 하나 더 추가했다.

파일 : task-controller.js

var Task = require('../models/task-model.js'); exports.list = function(req, res) { Task.find().sort('-priority').exec(function(err, tasks) { var planTasks = []; var todoTasks = []; var inProgressTasks = []; var doneTasks = []; for (var key in tasks) { var task = tasks[key]; if (task.get('status') === 'PLAN') { planTasks.push(task); } else if (task.get('status') === 'TO-DO') { todoTasks.push(task); } else if (task.get('status') === 'In-Progress') { inProgressTasks.push(task); } else if (task.get('status') === 'Done') { doneTasks.push(task); } else { console.error('태스크 상태가 유효하지 않습니다. tasks {' + tasks + '}'); } } res.render('index', { title: 'Kanban Project', planTasks: planTasks, todoTasks: todoTasks, inProgressTasks: inProgressTasks, doneTasks: doneTasks }); }); }; exports.create = function(req, res) { Task.find({ contents: req.body.contents }, function(err, tasks) { if (err) throw err; if (tasks.length > 0) { console.error('이미 같은 테스크가 존재합니다. {' + req.body.contents + '}'); } else { new Task({ contents: req.body.contents, priority: req.body.priority, author: req.body.author }).save(); console.log('새로운 테스크를 생성하였습니다. {' + req.body.contents + '}'); } }); res.redirect('/'); res.end(); }; exports.update = function(req, res) { Task.update({ contents: req.body.contents }, { status: req.body.status }, function(err, numberAffected, raw) { if (err) throw err; console.log('수정된 문서의 수는 %d 입니다.', numberAffected); console.log('MongoDB의 응답 : ', raw); }); res.redirect('/'); res.end(); }; exports.remove = function(req, res) { Task.remove({ contents: req.body.contents }, function(err) { if (err) throw err; console.log('태스크를 삭제하였습니다. {' + req.body.contents + '}'); }); res.redirect('/'); res.end(); };

​

상태에는 "PLAN"을 기본 값으로 설정하였고 우선순위에는 1을 기본값으로 설정하였다.

파일 : task-model.js

var mongoose = require('mongoose'); var taskSchema = new mongoose.Schema({ status: { type: String, default: "PLAN" }, priority: { type: int, default: 1 }, contents: { type: String, default: "" }, author: { type: String, default: "" } }); module.exports = mongoose.model("Task", taskSchema);
