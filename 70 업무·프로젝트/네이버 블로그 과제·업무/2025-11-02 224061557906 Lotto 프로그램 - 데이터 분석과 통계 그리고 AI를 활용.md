모든 가능한 조합

**전체 파일 생성**

""" 로또(6/45) 모든 조합 생성기 \- 조합 개수: C(45, 6) = 8,145,060 \- 메모리에 올리지 않고 스트리밍으로 파일에 기록 """ from itertools import combinations from math import comb from pathlib import Path import gzip import sys N = 45 # 번호 1..45 K = 6 # 6개 뽑기 def lotto_combinations(n: int = N, k: int = K): """오름차순 조합 튜플을 제너레이터로 스트리밍.""" return combinations(range(1, n + 1), k) def open_out(path: Path): """확장자로 출력 방식을 결정: .gz면 gzip 텍스트 모드, 아니면 일반 텍스트.""" if path.suffix == ".gz": return gzip.open(path, "wt", encoding="utf-8", newline="") return path.open("w", encoding="utf-8", newline="") def write_all(path: str = "lotto_6_45.csv.gz", progress_every: int = 200_000): """ 모든 조합을 path로 저장. \- path가 .gz로 끝나면 gzip 압축(csv.gz), 아니면 평문 csv. \- progress_every 마다 진행상황을 stderr로 출력(선택). """ out_path = Path(path) total = comb(N, K) # 8,145,060 print(f"쓰기 시작: {out_path} (예상 조합 수: {total:,})", file=sys.stderr) count = 0 with open_out(out_path) as f: # 헤더(원하면 제거 가능) f.write("n1,n2,n3,n4,n5,n6\n") for tup in lotto_combinations(N, K): # '1,4,17,23,35,44' 형태로 한 줄 기록 f.write(",".join(map(str, tup)) + "\n") count += 1 if progress_every and (count % progress_every == 0): print(f"{count:,}/{total:,} ({count/total:6.2%})", file=sys.stderr) print(f"완료: 총 {count:,}개 조합을 {out_path}에 저장했습니다.", file=sys.stderr) if __name__ == "__main__": # 평문 CSV write_all("data/lotto_6_45.csv")

**테이블 생성**

""" 로또(6/45) 모든 조합을 테이블에 저장하는 코드 C(45,6)=8,145,060 조합을 streaming으로 생성, batch insert로 저장 """ import pymysql from itertools import combinations from math import comb # ----------------------------- # 1. DB 연결 정보 # ----------------------------- DB_CONFIG = { "host": "127.0.0.1", "port": 3306, "user": "root", "password": "1234", "database": "thunder_lucky", "charset": "utf8mb4", } # ----------------------------- # 2. 테이블 생성 SQL # ----------------------------- CREATE_TABLE_SQL = """ CREATE TABLE IF NOT EXISTS lotto_6_45 ( id BIGINT AUTO_INCREMENT PRIMARY KEY, n1 TINYINT UNSIGNED NOT NULL, n2 TINYINT UNSIGNED NOT NULL, n3 TINYINT UNSIGNED NOT NULL, n4 TINYINT UNSIGNED NOT NULL, n5 TINYINT UNSIGNED NOT NULL, n6 TINYINT UNSIGNED NOT NULL, UNIQUE KEY uniq_nums (n1,n2,n3,n4,n5,n6) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4; """ # ----------------------------- # 3. 조합 생성기 # ----------------------------- def lotto_combinations(): for combo in combinations(range(1, 46), 6): yield combo # ----------------------------- # 4. 데이터베이스 연결 함수 # ----------------------------- def get_connection(): return pymysql.connect(**DB_CONFIG) # ----------------------------- # 5. 메인 로직 (배치 단위 INSERT) # ----------------------------- def insert_all(batch_size: int = 5000): total = comb(45, 6) conn = get_connection() cursor = conn.cursor() cursor.execute(CREATE_TABLE_SQL) conn.commit() sql = "INSERT IGNORE INTO lotto_6_45 (n1,n2,n3,n4,n5,n6) VALUES (%s,%s,%s,%s,%s,%s)" count = 0 batch = [] print(f"총 {total:,}개 조합 생성 및 저장 시작...") for combo in lotto_combinations(): batch.append(combo) if len(batch) >= batch_size: cursor.executemany(sql, batch) conn.commit() count += len(batch) print(f"{count:,}/{total:,} ({count/total:6.2%}) 완료") batch = [] # 남은 데이터 처리 if batch: cursor.executemany(sql, batch) conn.commit() count += len(batch) print(f"\n완료! 총 {count:,}개 조합이 MySQL에 저장되었습니다.") cursor.close() conn.close() # ----------------------------- # 실행 # ----------------------------- if __name__ == "__main__": insert_all(batch_size=5000)
