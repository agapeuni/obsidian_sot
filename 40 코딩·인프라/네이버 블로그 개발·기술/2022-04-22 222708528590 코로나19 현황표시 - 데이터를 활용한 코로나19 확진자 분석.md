4\. 예제 1

파이썬의 sum 함수와 numpy의 mean 함수를 사용하여 합계와 평균을 구해보자.

import numpy as np import matplotlib.pyplot as plt # 데이터 : 일자 d_date = ['7-01', '7-02', '7-03', '7-04', '7-05', '7-06', '7-07', '7-08', '7-09', '7-10', '7-11', '7-12', '7-13', '7-14', '7-15', '7-16', '7-17', '7-18', '7-19', '7-20', '7-21', '7-22', '7-23', '7-24', '7-25', '7-26', '7-27', '7-28', '7-29', '7-30'] # 데이터 : 확진자수 d_count = [761, 825, 794, 743, 711, 746, 1212, 1275, 1316, 1378, 1324, 1100, 1150, 1613, 1599, 1536, 1451, 1454, 1251, 1278, 1781, 1841, 1630, 1629, 1487, 1318, 1363, 1895, 1674, 1710] # 데이터 : 확진자수(국내) d_cnt_d = [711, 765, 748, 662, 644, 690, 1168, 1227, 1236, 1320, 1280, 1063, 1097, 1566, 1554, 1476, 1401, 1402, 1207, 1242, 1725, 1532, 1574, 1573, 1422, 1264, 1274, 1822, 1632, 1662] # 데이터 : 확진자수(해외) d_cnt_f = [50, 60, 46, 81, 67, 56, 44, 48, 80, 58, 44, 37, 53, 47, 45, 60, 50, 52, 44, 36, 56, 309, 56, 56, 65, 54, 89, 73, 42, 48] print("=== 7월 확진자수 ===") print("전체 합계", sum(d_count)) print("국내 합계", sum(d_cnt_d)) print("해외 합계", sum(d_cnt_f)) print("전체 일일평균", np.mean(d_count)) print("국내 일일평균", np.mean(d_cnt_d)) print("해외 일일평균", np.mean(d_cnt_f))

5\. 예제 2

준비된 데이터를 사용하여 matplotlib로 시각화한다.

import numpy as np import matplotlib.pyplot as plt # 데이터 : 일자 d_date = ['7-01', '7-02', '7-03', '7-04', '7-05', '7-06', '7-07', '7-08', '7-09', '7-10', '7-11', '7-12', '7-13', '7-14', '7-15', '7-16', '7-17', '7-18', '7-19', '7-20', '7-21', '7-22', '7-23', '7-24', '7-25', '7-26', '7-27', '7-28', '7-29', '7-30'] # 데이터 : 확진자수 d_count = [761, 825, 794, 743, 711, 746, 1212, 1275, 1316, 1378, 1324, 1100, 1150, 1613, 1599, 1536, 1451, 1454, 1251, 1278, 1781, 1841, 1630, 1629, 1487, 1318, 1363, 1895, 1674, 1710] # 데이터 : 확진자수(국내) d_cnt_d = [711, 765, 748, 662, 644, 690, 1168, 1227, 1236, 1320, 1280, 1063, 1097, 1566, 1554, 1476, 1401, 1402, 1207, 1242, 1725, 1532, 1574, 1573, 1422, 1264, 1274, 1822, 1632, 1662] # 데이터 : 확진자수(해외) d_cnt_f = [50, 60, 46, 81, 67, 56, 44, 48, 80, 58, 44, 37, 53, 47, 45, 60, 50, 52, 44, 36, 56, 309, 56, 56, 65, 54, 89, 73, 42, 48] # 한글 폰트 plt.rc('font', family='gulim') plt.title("코로나19 확진자 발생현황 (2021년 7월)") plt.xlabel("날짜") plt.ylabel("확진자수") plt.plot(d_date, d_count, label="합계") plt.plot(d_date, d_cnt_d, label="국내발생", c="blue") plt.plot(d_date, d_cnt_f, label="해외유입", c="green") plt.scatter(d_date, d_count, c="red") plt.legend() plt.show()
