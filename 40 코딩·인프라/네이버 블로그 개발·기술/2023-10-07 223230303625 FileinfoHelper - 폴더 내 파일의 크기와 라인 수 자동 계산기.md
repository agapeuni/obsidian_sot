프로젝트를 수행하고 있는 중에 고객으로부터 자료 요청을 받았다. 현재 운영서버에 서비스되고 있는 전체 Java 코드의 파일 크기와 라인수를 양식에 구애받지 말고 엑셀파일에 데이터로 정리해 달라는 내용이다.

​

Java 파일을 하나씩 확인하는 것은 무척 번거로운 작업이다. 그래서 해당 정보를 Console로 표시하는 Java 클래스를 만들었다. 확장자와 경로만 지정하면 파일 크기와 라인수를 파일명과 경로를 포함해 Tab으로 구분해서 출력한다. Tab으로 구분되어 있어 Console에 출력된 텍스트를 복사해서 엑셀에 바로 붙여넣기를 할 수 있다.

**▣ FileinfoHelper.java**

import java.io.BufferedReader; import java.io.File; import java.io.FileReader; import java.io.IOException; public class FileinfoHelper { private String extention; public FileinfoHelper(String extention) { this.extention = extention; } /** * 파일 사이즈 * * @param path * @return */ public long countTotalSize(String path) { long retval = 0L; File filepath = new File(path); File[] files = filepath.listFiles(); for (int i = 0, len = files.length; i < len; i++) { if (files[i].isDirectory()) { retval += countTotalSize(files[i].getPath()); } else if (files[i].isFile() && files[i].getName().toLowerCase().endsWith(extention)) { retval += files[i].length(); System.out.println(files[i].length() + "\t" + files[i].getName() + "\t" + files[i].getPath()); } } return retval; } /** * 파일 사이즈 * * @param path * @return */ public int countTotalLine(String path) { int retval = 0; File filepath = new File(path); File[] files = filepath.listFiles(); for (int i = 0, len = files.length; i < len; i++) { if (files[i].isDirectory()) { retval += countTotalLine(files[i].getPath()); } else if (files[i].isFile() && files[i].getName().toLowerCase().endsWith(extention)) { int lineCount = 0; try { BufferedReader in = new BufferedReader(new FileReader(files[i].getPath())); while (in.readLine() != null) { ++lineCount; } } catch (IOException ie) { ie.printStackTrace(); } retval += lineCount; System.out.println(lineCount + "\t" + files[i].getName() + "\t" + files[i].getPath()); } } return retval; } public static void main(String[] args) { // 파일정보를 가져올 확장자를 설정 String path = "D:\\\code\\\java\\\java-swing"; FileinfoHelper fileInfoHelper = new FileinfoHelper(".java"); System.out.println("Total Size = " + fileInfoHelper.countTotalSize(path)); // 파일의 크기 System.out.println("Total Line = " + fileInfoHelper.countTotalLine(path)); // 파일의 라인 } }
