**2\. CSV파일로 다운로드**

서울지역 1960년 1월부터 현재까지 기온자료를 CSV파일로 다운로드 하였다. 파일을 열어보면 1행부터 6행까지는 데이터에 대한 메타정보가 담겨있고 8행은 헤더정보이고 실제 데이터는 9행부터 위치한다. 데이터는 날짜, 지점, 평균기온과 최저기온 그리고 최고기온가 담겨있다.

**3\. 파이썬 코드 구현**

프로그램을 실행하기 전에 CSV 파일의 맨아래 빈라인을 먼저 제거한다. CSV 파일을 읽어 매년 8월 15일 평균기온, 최저기온, 최고기온을 각각의 리스트에 저장한다. CSV 파일의 모든 행을 반복하여 데이터 저장을 마치고 나면 matplotlib 라이브러리를 이용해 저장해둔 데이터를 기반으로 차트를 생성한다. 

import csv import matplotlib.pyplot as plt with open("ta_20220517215834.csv") as f: data = csv.reader(f) for i in range(8): next(data) high = [] avg = [] low = [] day = [] for row in data : if row[-1] != '' and row[-2] != '' : date = row[0].split('-') if date[1] == '08' and date[2] == '15' : high.append(float(row[-1])) low.append(float(row[-2])) avg.append(float(row[-3])) day.append(date[0]) plt.rc('font', family = 'Malgun Gothic') plt.rcParams['axes.unicode_minus'] = False plt.title('광복절날 기준 기온 변화') plt.plot(day, high, 'hotpink', label = '최고기온') plt.plot(day, low, 'skyblue', label = '최저기온') plt.plot(day, avg, 'darkgreen', label = '평균기온') plt.legend() plt.show()

**4\. 평균온도 확인**

다음은 2월, 5월, 8월, 11월의 평균온도에 대한 히스토그램을 생성하는 코드이다.

import csv import matplotlib.pyplot as plt with open("data/sample_1973.csv") as f: data = csv.reader(f) for i in range(8): next(data) m02 = [] m05 = [] m08 = [] m11 = [] for row in data : month = row[0].split('-')[1] if row[-1] != '' : if month == '02': m02.append(float(row[2])) if month == '05': m05.append(float(row[2])) if month == '08': m08.append(float(row[2])) if month == '11': m11.append(float(row[2])) plt.rc('font', family = 'Malgun Gothic') plt.rcParams['axes.unicode_minus'] = False plt.title('2월, 5월, 8월, 11월 평균기온') plt.hist(m11, bins = 100, color = 'gray', label = '11월') plt.hist(m08, bins = 100, color = 'r', label = '8월') plt.hist(m05, bins = 100, color = 'g', label = '4월') plt.hist(m02, bins = 100, color = 'b', label = '2월') plt.legend() plt.show()

import csv import matplotlib.pyplot as plt with open("ta_20220517215834.csv") as f: data = csv.reader(f) for i in range(8): next(data) m02 = [] m05 = [] m08 = [] m11 = [] for row in data : year = int(row[0].split('-')[0]) month = row[0].split('-')[1] if row[-1] != '' : # 2001년부터 2022년 데이터를 필터 if 2000 < year and year <= 2022: if month == '02': m02.append(float(row[2])) if month == '05': m05.append(float(row[2])) if month == '08': m08.append(float(row[2])) if month == '11': m11.append(float(row[2])) plt.rc('font', family = 'Malgun Gothic') plt.rcParams['axes.unicode_minus'] = False plt.title('2001년부터 2022년까지 2월, 5월, 8월, 11월 평균기온') plt.hist(m02, bins = 100, color = 'b', label = '2월') plt.hist(m05, bins = 100, color = 'g', label = '4월') plt.hist(m08, bins = 100, color = 'r', label = '8월') plt.hist(m11, bins = 100, color = 'gray', label = '11월') plt.legend() plt.show()
