1\. Java Swing 개요

Java Swing은 자바 플랫폼에서 GUI(Graphical User Interface)를 개발하기 위한 툴킷이다. Swing은 자바의 표준 라이브러리인 Java Foundation Classes (JFC)의 일부로 제공된다. Swing은 AWT(Abstract Window Toolkit)의 후속 버전으로 개발되었다. AWT는 Java의 초기 GUI 툴킷이었지만, 플랫폼 독립성과 다양한 컴포넌트를 지원하기에는 제한적이었다. Swing은 AWT의 한계를 극복하고 풍부한 기능과 컴포넌트를 제공하는 목적으로 개발되었다.

​

Swing은 다양한 컴포넌트를 제공하여 사용자 인터페이스를 구성할 수 있다. 버튼, 레이블, 텍스트 필드, 체크 박스, 라디오 버튼, 리스트 박스, 테이블 등 다양한 컴포넌트를 포함하고 있다. 이러한 컴포넌트들을 사용하여 사용자 인터페이스를 구성하고, 이벤트 처리 및 액션을 관리할 수 있다.

​

Swing은 플랫폼 독립적이며, JVM 상에서 실행된다. 이는 개발자가 작성한 코드가 서로 다른 운영체제에서 동일하게 동작할 수 있도록 한다. Swing은 다양한 외관을 지원하며, 개발자가 컴포넌트의 스타일과 외관을 쉽게 변경할 수 있다. Java Swing은 자바 개발자들에게 다양한 GUI 애플리케이션 개발의 가능성을 제공하며, 쉬운 사용법과 풍부한 기능을 가지고 있다.

​

Swing은 이벤트 기반 프로그래밍 모델을 사용하여 사용자와의 상호작용을 처리한다. 예를 들어, 버튼 클릭, 텍스트 입력 등의 이벤트가 발생하면 해당 이벤트를 처리하는 코드를 작성할 수 있다. Swing은 이벤트 처리를 위한 다양한 인터페이스와 클래스를 제공하여 이를 지원한다.

2\. Hello World 예제

Swing을 이용하여 Hello World를 만들어 보았다. Hello World는 프로그래밍 언어를 배울 때 처음 만들어보는 예제이다. 

화면에는 입력항목과 버튼을 있어 버튼을 클릭할 때 "Hello Swing. *^^*"를 출력한다. 

▣ HelloSwing.java

import java.awt.BorderLayout; import java.awt.event.ActionEvent; import java.awt.event.ActionListener; import javax.swing.JButton; import javax.swing.JFrame; import javax.swing.JLabel; import javax.swing.JPanel; import javax.swing.JTextField; public class HelloSwing implements ActionListener { JFrame frame = new JFrame("Hello App"); JPanel panel1 = new JPanel(); JPanel panel2 = new JPanel(); JPanel panel3 = new JPanel(); JLabel label = new JLabel("Hello Label"); JButton button = new JButton("Click Me"); JTextField textField = new JTextField(10); public void start() { panel1.add(label); frame.getContentPane().add(BorderLayout.NORTH, panel1); panel2.add(textField); frame.getContentPane().add(BorderLayout.CENTER, panel2); button.addActionListener(this); panel3.add(button); frame.getContentPane().add(BorderLayout.SOUTH, panel3); frame.setSize(300, 200); frame.setVisible(true); } public void actionPerformed(ActionEvent e) { if (button == e.getSource()) { textField.setText("Hello Swing"); } } public static void main(String[] args) { HelloSwing hello = new HelloSwing(); hello.start(); } }

항목별로 코드를 정리했다. 동작하는 기능은 동일하다.

import java.awt.BorderLayout; import javax.swing.JButton; import javax.swing.JFrame; import javax.swing.JLabel; import javax.swing.JPanel; import javax.swing.JTextField; public class HelloSwingEx extends JFrame { private static final long serialVersionUID = 1L; private JPanel jContentPane = null; private JPanel panel1 = null; private JPanel panel2 = null; private JPanel panel3 = null; private JLabel jLabel = null; private JButton jButton = null; private JTextField jTextField = null; public static void main(String[] args) { new HelloSwingEx(); } public HelloSwingEx() { initialize(); setVisible(true); } private void initialize() { this.setSize(300, 200); this.setLocation(100, 100); this.setContentPane(getJContentPane()); this.setTitle("Hello Swing"); this.setDefaultCloseOperation(EXIT_ON_CLOSE); } private JPanel getJContentPane() { if (jContentPane == null) { jContentPane = new JPanel(); jContentPane.setLayout(new BorderLayout()); panel1 = new JPanel(); panel2 = new JPanel(); panel3 = new JPanel(); jLabel = new JLabel("Hello Label"); panel1.add(jLabel); panel2.add(getJTextField()); panel3.add(getJButton()); jContentPane.add(BorderLayout.NORTH, panel1); jContentPane.add(BorderLayout.CENTER, panel2); jContentPane.add(BorderLayout.SOUTH, panel3); } return jContentPane; } private JTextField getJTextField() { if (jTextField == null) { jTextField = new JTextField(10); } return jTextField; } private JButton getJButton() { if (jButton == null) { jButton = new JButton(); jButton.setText("Click Me"); jButton.addActionListener(new java.awt.event.ActionListener() { public void actionPerformed(java.awt.event.ActionEvent e) { jTextField.setText("Hello Swing"); } }); } return jButton; } }
