아래의 쿼리에서 **'TABLENAME_HERE_%'** 부분을 프로젝트 상황에 따라 적절히 입력하면 된다.

​

**▣ TABLE_LIST_DEF.sql**

SELECT A1.TABLE_COMMENTS ,A1.TABLE_NAME ,A1.COLUMN_ID ,A1.COLUMN_COMMENTS ,A1.COLUMN_NAME ,A1.DATA_TYPE ,A1.DATA_LENGTH ,A1.DATA_SCALE ,DECODE(B1.CONSTRAINT_TYPE, 'P', 'O') PK_FLAG ,DECODE(B1.CONSTRAINT_TYPE, 'P', '', A1.NULL_FLAG) NULL_FLAG ,A1.DATA_DEFAULT FROM (SELECT B.COMMENTS TABLE_COMMENTS ,A.TABLE_NAME TABLE_NAME ,A.COLUMN_ID ,C.COMMENTS COLUMN_COMMENTS ,A.COLUMN_NAME COLUMN_NAME ,A.DATA_TYPE ,(CASE WHEN A.DATA_TYPE IN ('CHAR', 'VARCHAR2') THEN A.DATA_LENGTH WHEN A.DATA_TYPE = 'NUMBER' AND A.DATA_PRECISION IS NOT NULL THEN A.DATA_PRECISION END) DATA_LENGTH ,A.DATA_SCALE ,DECODE(A.NULLABLE, 'N', 'X') NULL_FLAG ,A.DATA_DEFAULT FROM USER_TAB_COLUMNS A JOIN USER_TAB_COMMENTS B ON (A.TABLE_NAME = B.TABLE_NAME) JOIN USER_COL_COMMENTS C ON (A.TABLE_NAME = C.TABLE_NAME AND A.COLUMN_NAME = C.COLUMN_NAME) WHERE A.TABLE_NAME LIKE 'TABLENAME_HERE_%') A1 LEFT JOIN (SELECT A.TABLE_NAME, A.COLUMN_NAME, B.CONSTRAINT_TYPE FROM USER_CONS_COLUMNS A, USER_CONSTRAINTS B WHERE (A.CONSTRAINT_NAME = B.CONSTRAINT_NAME) AND B.CONSTRAINT_TYPE IN ('P')) B1 ON (A1.TABLE_NAME = B1.TABLE_NAME AND A1.COLUMN_NAME = B1.COLUMN_NAME) ORDER BY A1.TABLE_NAME, A1.COLUMN_ID
