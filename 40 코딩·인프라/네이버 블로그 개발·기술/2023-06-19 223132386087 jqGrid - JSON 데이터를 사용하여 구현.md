1\. jqGrid 개요

jqGrid는 jQuery를 기반으로 한 오픈 소스 JavaScript 그리드 플러그인이다. jqGrid는 웹에서 테이블 형식 데이터를 표현하고 조작하기 위한 JavaScript 컨트롤이다. jqGrid 5.8.2 버전이 2023년 1월 30일에 출시했다. 라이선스 정책이 변경되어서 4.7 이하의 버전은 무료이고 4.8 이상의 버전은 유료이다. 30일 평가판으로 미리 사용해 볼 수 있다. 

​

JSON 데이터를 사용하여 jqGrid를 구현했다. USER 테이블에서 데이터를 읽어와 JSON 형태로 만들고 JSON 문자열을 JavaScript에서 eval 함수를 사용하여 json 객체로 만들어 jqGrid 데이터로 추가한다. 예제 수준의 코드라 다소 거친 부분이 있는 점 양해를 바란다. 

​

3\. Java 소스

USER 테이블을 조회하여 Row 데이터를 한 줄씩 JSON에 key/value 형태로 저장한다.

import java.sql.Connection; import java.sql.PreparedStatement; import java.sql.ResultSet; import java.sql.SQLException; import com.TssConnectionResource; import net.sf.json.util.JSONStringer; public class GridDao { public JSONStringer getJson() { Connection con = null; ResultSet rset = null; PreparedStatement pstmt = null; TssConnectionResource tssCon = null; JSONStringer json = null; StringBuffer query = new StringBuffer(); query.append(" SELECT * FROM USER "); try { tssCon = new TssConnectionResource(); con = tssCon.getConnection(); pstmt = con.prepareStatement(query.toString()); rset = pstmt.executeQuery(); json = new JSONStringer(); json.array(); while (rset.next()) { json.object() .key("USERIDNO").value(rset.getString("USERIDNO")) .key("USERNAME").value(rset.getString("USERNAME")) .key("USERDETP").value(rset.getString("USERDETP")) .key("USERGRDE").value(rset.getString("USERGRDE")) .key("PHONNUMB").value(rset.getString("PHONNUMB")) .key("USERMAIL").value(rset.getString("USERMAIL")) .key("USERPERM").value(rset.getString("USERPERM")).endObject(); } json.endArray(); } catch (SQLException e) { System.out.println(e.toString()); } finally { try { if (rset != null) rset.close(); if (pstmt != null) pstmt.close(); if (tssCon != null) tssCon.close(); } catch (SQLException se) { System.out.println(se.toString()); } } return json; } }
