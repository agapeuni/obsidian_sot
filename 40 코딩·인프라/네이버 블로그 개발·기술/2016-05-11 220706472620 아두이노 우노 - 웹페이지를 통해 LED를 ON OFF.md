**▣ 소스코드 : ethernet_2_onoff.ino**

#include <Ethernet.h> #include <SPI.h> byte MAC_ADRS[] = { 0xB0, 0x12, 0xFA, 0x34, 0x1A, 0xE2 }; EthernetServer ethernetServer = EthernetServer(80); int pin = 2; void setup() { Serial.begin(9600); pinMode(pin, OUTPUT); if (!Ethernet.begin(MAC_ADRS)) { Serial.println("Ethernet.begin() NG"); return; } else { Serial.println("Ethernet.begin() OK"); } ethernetServer.begin(); Serial.println("Server Started..."); Serial.print("IP: "); Serial.println(Ethernet.localIP()); } void loop() { EthernetClient ethernetClient = ethernetServer.available(); if (ethernetClient) { Serial.println("EthernetClient connected"); boolean IsBlankLine = true; String buffer = ""; while (ethernetClient.connected()) { if (ethernetClient.available()) { char c = ethernetClient.read(); buffer += c; Serial.write(c); if (c == '\n' && IsBlankLine) { ethernetClient.println("HTTP/1.1 200 OK"); ethernetClient.println("Content-Type: text/html"); ethernetClient.println(); ethernetClient.println("<form method=\"get\" action=\"/led.do\">"); ethernetClient.println("<input type='hidden' name='cmd' value='0' />"); ethernetClient.println("<input type='submit' value='LED Off' />"); ethernetClient.println("</form>"); ethernetClient.println("<form method=\"get\" action=\"/led.do\">"); ethernetClient.println("<input type='hidden' name='cmd' value='1' />"); ethernetClient.println("<input type='submit' value='LED On' />"); ethernetClient.println("</form>"); break; } if (c == '\n') { IsBlankLine = true; buffer = ""; } else if (c == '\r') { if (buffer.indexOf("GET /led.do?cmd=0") >= 0) { digitalWrite(pin, LOW); ethernetClient.println("HTTP/1.1 200 OK"); ethernetClient.println("Content-Type: text/html"); ethernetClient.println(); ethernetClient.println("LED is Off"); ethernetClient.println("<br />"); ethernetClient.println("<form method=\"get\" action=\"/led.do\">"); ethernetClient.println("<input type='hidden' name='cmd' value='1' />"); ethernetClient.println("<input type='submit' value='LED On' />"); ethernetClient.println("</form>"); IsBlankLine = false; break; } if (buffer.indexOf("GET /led.do?cmd=1") >= 0) { digitalWrite(pin, HIGH); ethernetClient.println("HTTP/1.1 200 OK"); ethernetClient.println("Content-Type: text/html"); ethernetClient.println(); ethernetClient.println("LED is On"); ethernetClient.println("<br />"); ethernetClient.println("<form method=\"get\" action=\"/led.do\">"); ethernetClient.println("<input type='hidden' name='cmd' value='0' />"); ethernetClient.println("<input type='submit' value='LED Off' />"); ethernetClient.println("</form>"); IsBlankLine = false; break; } } else { IsBlankLine = false; } } } delay(5); ethernetClient.stop(); Serial.println("\nEthernetClient disonnected"); } }

#include <Ethernet.h> #include <SPI.h> byte MAC_ADRS[] = { 0xB0, 0x12, 0xFA, 0x34, 0x1A, 0xE2 }; EthernetServer ethernetServer(80); IPAddress ip(192, 168, 0, 7); IPAddress gateway(192, 168, 0, 1); IPAddress subnet(255, 255, 255, 0); int pin2 = 2; int pin3 = 3; int pin4 = 4; void setup() { Serial.begin(9600); pinMode(pin2, OUTPUT); pinMode(pin3, OUTPUT); pinMode(pin4, OUTPUT); if (!Ethernet.begin(MAC_ADRS)) { Serial.println("Ethernet.begin() NG"); return; } else { Serial.println("Ethernet.begin() OK"); } ethernetServer.begin(); Serial.println("Server Started..."); Serial.print("IP: "); Serial.println(Ethernet.localIP()); } void loop() { EthernetClient ethernetClient = ethernetServer.available(); if (ethernetClient) { Serial.println("EthernetClient connected"); boolean IsBlankLine = true; String buffer = ""; while (ethernetClient.connected()) { if (ethernetClient.available()) { char c = ethernetClient.read(); buffer += c; Serial.write(c); if (c == '\n' && IsBlankLine) { ethernetClient.println("HTTP/1.1 200 OK"); ethernetClient.println("Content-Type: text/html"); ethernetClient.println(); ethernetClient.println("<html>"); ethernetClient.println("<body>"); if (digitalRead(pin2) > 0) { ethernetClient.println("LED is Pin 2"); } else if (digitalRead(pin3) > 0) { ethernetClient.println("LED is Pin 3"); } else if (digitalRead(pin4) > 0) { ethernetClient.println("LED is Pin 4"); } else { ethernetClient.println("LED is Off"); } ethernetClient.println("<br />"); ethernetClient.println("<form method=\"get\" action=\"/led.do\">"); ethernetClient.println("<input type='hidden' name='on' value='2' />"); ethernetClient.println("<input type='submit' value='LED Pin 3' />"); ethernetClient.println("</form>"); ethernetClient.println("<form method=\"get\" action=\"/led.do\">"); ethernetClient.println("<input type='hidden' name='on' value='4' />"); ethernetClient.println("<input type='submit' value='LED Off' />"); ethernetClient.println("</form>"); ethernetClient.println("</body>"); ethernetClient.println("</html>"); break; } 

if (c == '\n') { IsBlankLine = true; buffer = ""; } else if (c == '\r') { if (buffer.indexOf("GET /led.do?on=0") >= 0) { digitalWrite(pin2, LOW); digitalWrite(pin3, LOW); digitalWrite(pin4, LOW); } if (buffer.indexOf("GET /led.do?on=2") >= 0) { digitalWrite(pin2, HIGH); digitalWrite(pin3, LOW); digitalWrite(pin4, LOW); } if (buffer.indexOf("GET /led.do?on=3") >= 0) { digitalWrite(pin2, LOW); digitalWrite(pin3, HIGH); digitalWrite(pin4, LOW); } if (buffer.indexOf("GET /led.do?on=4") >= 0) { digitalWrite(pin2, LOW); digitalWrite(pin3, LOW); digitalWrite(pin4, HIGH); } if (buffer.indexOf("GET /led.do?on=") >= 0) { ethernetClient.println("HTTP/1.1 200 OK"); ethernetClient.println("Content-Type: text/html\n"); ethernetClient.println();
