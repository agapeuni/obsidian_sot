​

요청 파라미터를 파싱 하여 화면에 출력하는 예제다.

var express = require('express'); var url = require('url'); var ejs = require('ejs'); var app = express(); app.set('views', __dirname + '/views'); app.set('view engine', 'html'); app.engine('html', ejs.renderFile); app.get('/', function(req, res) { res.render('index.html'); }); app.get('/member', function(req, res) { var urlParse = url.parse(req.url, true); var queryString = urlParse.query; if (queryString.name) { res.send('Your name is ' + queryString.name); } else { res.send('Input your name'); } }); app.get('/member/:id', function(req, res) { res.send('member= ' + req.param("id")); }); var server = app.listen(3000, function() { console.log("http://localhost:3000/"); });

​

먼저 URL 모듈을 추가한다. URL 모듈로 파라미터를 파싱 하여 요청 파라미터를 처리한다.

var url = require('url');  
---  
  
​

'/member'의 요청에 대해서 name 파라미터의 값을 출력하는 코드이다.

app.get('/member', function(req, res) { var urlParse = url.parse(req.url, true); var queryString = urlParse.query; res.send('Your name is ' + queryString.name); });

​

[http://localhost:3000/member? name=One Way를](http://localhost:3000/member?name=OneWay) 브라우저에 입력한다. url 모듈의 parse() 메서드를 사용하여 파싱 한다.

결국에 쿼리 스트링 참조변수를 얻게 되는데 querystring.name으로 값 'Express'를 가져올 수 있다.

![](https://postfiles.pstatic.net/MjAyMzA5MDFfMjEg/MDAxNjkzNTM5NTI5NjU2.3nlqLru5ZytvHxJX5rzHAAUhrkQ3SKB1H6U7QvfaSvEg.TT2sAmnXOu9-Psr775raQ0vOF1zibJ9w0zZBT9b5Xtwg.PNG.agapeuni/img.png?type=w80_blur)

​

'<http://localhost:3000/member>'와 같이 파라미터 없이 주소를 입력하면 queryString.name에 할당된 값이 없기 때문에 'undefined'라고 표시된다.

![](https://postfiles.pstatic.net/MjAyMzA5MDFfMTgg/MDAxNjkzNTM5NTM0Mjcz.7QMDusObiM2DX4N2yyg0Crlyx_P-tal186cCiWOwdDgg.eG1w5Ays8GejAXOO6yOyFyS71DGYhpZWdqeIpzbuqEUg.PNG.agapeuni/img.png?type=w80_blur)

​

if 조건문에 queyrString.name의 값에 따라 분기를 한다. 그러면 값이 있을 때는 출력하고 값이 없을 때(undefined인 경우)에 else 문이 실행되어 이름을 입력하라는 메시지가 표시된다.

app.get('/member', function(req, res) { var urlParse = url.parse(req.url, true); var queryString = urlParse.query; if (queryString.name) { res.send('Your name is ' + queryString.name); } else { res.send('Input your name'); } });

다음은 RESTful 방식의 URL을 파싱 하여 화면에 출력하는 코드이다.

app.get('/member/:id', function(req, res) { res.send('member= ' + req.param("id")); });

​

브라우저에 [http://localhost:3000/member/one way를](http://localhost:3000/member/oneway) 입력한다.

/member/ 아래의 입력을 :id에 매핑하여 req.param("id")로 값을 받아올 수 있다.
