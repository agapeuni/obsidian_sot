**2\. web.xml 에 추가**

web.xml 파일에 JFreeChart 서블릿 관련 설정을 추가한다.

<servlet> <servlet-name>DisplayChart</servlet-name> <servlet-class>org.jfree.chart.servlet.DisplayChart</servlet-class> </servlet> <servlet-mapping> <servlet-name>DisplayChart</servlet-name> <url-pattern>/servlet/DisplayChart</url-pattern> </servlet-mapping>

​

​

**3\. JSP 작성**

차트에 표시되는 데이터를 JSP에서 생성한다. 

테스트를 위한 목적으로 소스코드가 클린하지 않다.

<%@ page contentType="text/html;charset=euc-kr"%> <%@ page import="org.jfree.data.general.DefaultPieDataset"%> <%@ page import="org.jfree.chart.JFreeChart"%> <%@ page import="org.jfree.chart.plot.PiePlot"%> <%@ page import="org.jfree.chart.ChartRenderingInfo"%> <%@ page import="org.jfree.chart.servlet.ServletUtilities"%> <%@ page import="org.jfree.chart.urls.StandardPieURLGenerator"%> <%@ page import="org.jfree.chart.entity.StandardEntityCollection"%> <%@ page import="org.jfree.chart.ChartFactory"%> <%@ page import="org.jfree.chart.ChartUtilities"%> <%@ page import="org.jfree.data.general.PieDataset"%> <%@ page import="org.jfree.data.category.DefaultCategoryDataset"%> <%@ page import="org.jfree.chart.plot.PlotOrientation"%> <%@ page import="org.jfree.chart.servlet.*"%> <%@ page import="java.io.*"%> <% String[] arr_data = request.getParameterValues("arr_data"); // 데이터 String[] arr_title = request.getParameterValues("arr_title"); // 제목 String chart_title = request.getParameter("chart_title") == null ? "" : cmUtil.mAscToKsc(request.getParameter("chart_title")); // 차트 제목 String chart_y_title = request.getParameter("chart_y_title") == null ? "" : request.getParameter("chart_y_title"); // y축 제목 String chart_x_title = request.getParameter("chart_x_title") == null ? "" : request.getParameter("chart_x_title"); // x축 제목 int i = 0; DefaultCategoryDataset dataset = new DefaultCategoryDataset(); if(arr_data !=null ) { for(i=0; i < arr_data.length; i++) { //차트에 보여질 데이터를 차트에 넣음. dataset.addValue(Double.parseDouble(arr_data[i]), chart_x_title, cmUtil.mAscToKsc(arr_title[i])); } } try { //막대 차트 JFreeChart chart = ChartFactory.createBarChart("Bar Chart","", chart_y_title, dataset, PlotOrientation.VERTICAL, true, true, false); chart.setBackgroundPaint(java.awt.Color.white); chart.setTitle(chart_title); ChartRenderingInfo info = new ChartRenderingInfo(new StandardEntityCollection()); //특정 임시 폴더에 이미지 파일을 만들어냄. String fileName = ServletUtilities.saveChartAsPNG(chart, 600, 300, info, session); String graphURL = request.getContextPath() + "/servlet/DisplayChart?filename=" + fileName; %> <img src="<%=graphURL%>"/> <!--//파일 경로와 파일명을 받아서 차트를 보여줌--> <br><br> <% //파이 차트 int total = 0; DefaultPieDataset dataset2 = new DefaultPieDataset(); if(arr_data !=null ) { for(i=0; i < arr_data.length; i++) { total += Integer.parseInt(arr_data[i]); } for(i=0; i < arr_data.length; i++) { dataset2.setValue(cmUtil.mAscToKsc(arr_title[i]), (Double.parseDouble(arr_data[i])/total)*100); } } JFreeChart chart2 = ChartFactory.createPieChart("Pie Chart", dataset2, true, true, false); chart2.setBackgroundPaint(java.awt.Color.white); chart2.setTitle(chart_title); ChartRenderingInfo info2 = new ChartRenderingInfo(new StandardEntityCollection()); String fileName2 = ServletUtilities.saveChartAsPNG(chart2, 600, 300, info, session); graphURL = request.getContextPath() + "/servlet/DisplayChart?filename=" + fileName2; %> <img src="<%=graphURL%>"/> <% } catch (Exception e) { System.out.println("error!! n"+e.toString()); } %>
