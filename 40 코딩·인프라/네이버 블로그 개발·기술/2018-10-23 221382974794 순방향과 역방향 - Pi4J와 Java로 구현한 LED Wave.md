이번에는 Pi4J 라이브러리를 사용하여 여러개의 LED로 Wave 효과를 구현해 보겠습니다.

​

아래는 이번 실험을 위한 개발환경입니다.

  * 라즈베리 파이 1 모델 B (Rev 2)

  * Linux raspberrypi 4.14.71+

  * java "1.8.0_65"

  * Pi4J 1.2-SNAPSHOT

​

매번 핀을 눈으로 확인하는 것이 쉽지 않아 라즈베리 파이와 브레드보드를 늘솜 확장보드로 연결하였습니다.

6개 핀을 정의하고 setup() 메소드에서 선언된 핀을 디지털 출력으로 설정합니다. 

loop() 메소드에서 LED에 대한 순차적인 On/Off를 구현합니다.

package com.pi4j.book.ch06; import static com.pi4j.util.Pi4jUtil.delay; import com.pi4j.io.gpio.GpioController; import com.pi4j.io.gpio.GpioFactory; import com.pi4j.io.gpio.GpioPinDigitalOutput; import com.pi4j.io.gpio.Pin; import com.pi4j.io.gpio.PinState; import com.pi4j.io.gpio.RaspiPin; public class S0603_MultiWave_1 { // 지연시간 int delayTime = 200; // 핀 정의 private static final Pin GPIO01 = RaspiPin.GPIO_01; private static final Pin GPIO02 = RaspiPin.GPIO_02; private static final Pin GPIO03 = RaspiPin.GPIO_03; private static final Pin GPIO04 = RaspiPin.GPIO_04; private static final Pin GPIO05 = RaspiPin.GPIO_05; private static final Pin GPIO06 = RaspiPin.GPIO_06; // 핀 배열 선언 final Pin[] pinArray = { GPIO01, GPIO02, GPIO03, GPIO04, GPIO05, GPIO06 }; final int pinLength = pinArray.length; GpioController gpioController = GpioFactory.getInstance(); GpioPinDigitalOutput[] pins = new GpioPinDigitalOutput[pinLength]; void setup() { for (int i = 0; i < pinArray.length; i++) { pins[i] = gpioController.provisionDigitalOutputPin(pinArray[i], PinState.LOW); } } void loop() { // 좌측에서 하나씩 순차적으로 On for (int i = 0; i < pinLength; i++) { pins[i].high(); delay(delayTime); } // 좌측에서 하나씩 순차적으로 Off for (int i = 0; i < pinLength; i++) { pins[i].low(); delay(delayTime); } } void shutdown() { gpioController.shutdown(); } public static void main(String[] args) { S0603_MultiWave_1 instance = new S0603_MultiWave_1(); instance.setup(); for (int i = 0; i < 7; i++) { System.out.println("Count" + i); instance.loop(); } instance.shutdown(); } }

소스코드 S0603_MultiWave_1 클래스를 실행한 동영상입니다.

​

​

아래의 코드는 위의 코드와 loop 메소드에서 LED 순차적인 Off일때만 코드가 다릅니다. 

위의 코드와 다르게 역순으로 LED 가 Off 됩니다.

package com.pi4j.book.ch06; import static com.pi4j.util.Pi4jUtil.delay; import com.pi4j.io.gpio.GpioController; import com.pi4j.io.gpio.GpioFactory; import com.pi4j.io.gpio.GpioPinDigitalOutput; import com.pi4j.io.gpio.Pin; import com.pi4j.io.gpio.PinState; import com.pi4j.io.gpio.RaspiPin; public class S0603_MultiWave_2 { // 지연시간 int delayTime = 200; // 핀 정의 private static final Pin GPIO01 = RaspiPin.GPIO_01; private static final Pin GPIO02 = RaspiPin.GPIO_02; private static final Pin GPIO03 = RaspiPin.GPIO_03; private static final Pin GPIO04 = RaspiPin.GPIO_04; private static final Pin GPIO05 = RaspiPin.GPIO_05; private static final Pin GPIO06 = RaspiPin.GPIO_06; // 핀 배열 선언 final Pin[] pinArray = { GPIO01, GPIO02, GPIO03, GPIO04, GPIO05, GPIO06 }; final int pinLength = pinArray.length; // 디지털 출력핀 선언 GpioController gpioController = GpioFactory.getInstance(); GpioPinDigitalOutput[] pins = new GpioPinDigitalOutput[pinLength]; void setup() { for (int i = 0; i < pinArray.length; i++) { pins[i] = gpioController.provisionDigitalOutputPin(pinArray[i], PinState.LOW); } } void loop() { // 좌측에서 하나씩 순차적으로 On for (int i = 0; i < pinLength; i++) { pins[i].high(); delay(delayTime); } // 우측에서 하나씩 순차적으로 Off for (int i = pinLength - 1; i > -1; i--) { pins[i].low(); delay(delayTime); } } void shutdown() { gpioController.shutdown(); } public static void main(String[] args) { S0603_MultiWave_2 instance = new S0603_MultiWave_2(); instance.setup(); for (int i = 0; i < 7; i++) { System.out.println("Count" + i); instance.loop(); } instance.shutdown(); } }
