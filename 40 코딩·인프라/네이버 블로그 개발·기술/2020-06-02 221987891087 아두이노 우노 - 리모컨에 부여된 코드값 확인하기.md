다음 스케치를 업로드하여 리모컨 버튼에 부여된 코드를 시리얼 모니터에서 확인해 봅니다.

**소스 코드 : ir_1_decode.ino**

#include <IRremote.h> int RECV_PIN = 3; IRrecv irrecv(RECV_PIN); decode_results results; void setup() { Serial.begin(9600); irrecv.enableIRIn(); } void loop() { // 리모콘 값을 입력받아 시리얼로 출력 if (irrecv.decode(&results)) { Serial.println(results.value, HEX); irrecv.resume(); } delay(100); }

​

리모컨 버튼을 클릭하면 각각의 코드값이 출력됩니다. 시리얼 모니터에서 값을 복사해 두어 사용하면 됩니다.

리모컨에서 확인한 코드로 아래와 같이 switch로 구분하면 각각의 기능을 수행할 수 있습니다. 여기서는 확인을 위해 LED를 On/Off 하였습니다.

​

**소스 코드 : ir_2_led_on.ino**

#include <IRremote.h> int RECV_PIN = 3; IRrecv irrecv(RECV_PIN); decode_results results; int pins[] = { 4, 5, 6, 7, 8, 9 }; int pinLen = 6; void setup() { irrecv.enableIRIn(); for (int i = 0; i < pinLen; i++) { pinMode(pins[i], OUTPUT); } } void loop() { int idx = -1; if (irrecv.decode(&results)) { switch (results.value) { case 0xFFC03F: idx = 0; break; case 0xFF20DF: idx = 1; break; case 0xFFA05F: idx = 2; break; case 0xFF00FF: idx = 3; break; case 0xFF807F: idx = 4; break; case 0xFF40BF: idx = 5; break; } ledOn(pins[idx]); irrecv.resume(); } delay(100); } void ledOn(int pin) { digitalWrite(pin, HIGH); delay(500); digitalWrite(pin, LOW); }
