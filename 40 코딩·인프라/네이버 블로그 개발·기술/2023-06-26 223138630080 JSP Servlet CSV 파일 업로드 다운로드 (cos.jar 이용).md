5\. 다운로드 (download.jsp)

메인페이지에서 전달한 파일이름과 시스템일시를 더하여 다운로드할 파일이름을 생성한다.

CSV 파일에는 Header를 작성하고 2건의 Data를 작성한다.

<%@ page contentType="application/csv; charset=UTF-8"%><%@ page import="java.util.Date"%><%@ page import="java.text.SimpleDateFormat"%> <% String CSV_DELIMITER = ","; String filename = request.getParameter("txt_name") == null ? "" : request.getParameter("txt_name").trim(); SimpleDateFormat sdf = new SimpleDateFormat("yyyyMMdd_HHmmss"); String timestamp = sdf.format(new Date()); response.setHeader("Content-Description", "CSV Generated Data"); response.setHeader("Content-Disposition", "attachment; filename=" + filename + "_" + timestamp + ".csv"); // CSV Header out.print("아이디" + CSV_DELIMITER); out.print("비번" + CSV_DELIMITER); out.print("이름" + CSV_DELIMITER); out.print("비고"); out.println(""); // CSV Data 1 out.print("ADMIN" + CSV_DELIMITER); out.print("1234" + CSV_DELIMITER); out.print("관리자" + CSV_DELIMITER); out.print("Hello CSV."); out.println(""); // CSV Data 2 out.print("TEST" + CSV_DELIMITER); out.print("1234" + CSV_DELIMITER); out.print("개발자" + CSV_DELIMITER); out.print("Hi CSV."); out.println(""); %>
