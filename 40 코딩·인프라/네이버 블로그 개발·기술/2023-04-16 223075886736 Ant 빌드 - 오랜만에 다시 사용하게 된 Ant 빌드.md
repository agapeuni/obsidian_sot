1\. Ant 빌드

진행하고 있는 프로젝트에서 Ant 빌드를 사용하게 되었다. 컴파일과 패키지는 Maven에서 진행하고 애플리케이션 환경 파일과 생성된 WAR 파일을 웹서버로 배포하는 과정을 Ant 빌드로 구현했다. 사용한 지 오래된 기술이지만 표현의 명료함과 구현의 자유도로 인해 선택했다. 

​

3\. 예제

웹 프로젝트를 진행하게 되면 코드를 수정하고 톰캣에 파일을 복사하고 실행하는 등의 많은 작업이 필요하다. 소스를 변경할 때마다 매번 같은 작업을 반복해야 한다. Ant 빌드를 사용하면 반복적인 작업을 단순화할 수 있다. 필요한 과정을 순서에 맞게 기술하여 단계별로 작업을 실행시켜서 마지막 ”웹 배포”까지 완료한다. 

​

다음은 Ant 프로젝트의 기본 파일이다. 

<project> 태그 안에 필요한 <target>을 정의한다.

​

**project**

<project name="WebApp" default="all"> <property environment="env"/> <property file="build.properties"/> <property file="${resource.dir}/dbpool.properties"/> <target name="init> ... (아래 내용 참조) ... </target> <target name="all"> <antcall target="clean" /> <antcall target="resources" /> <antcall target="copy" /> <antcall target="compile" /> <antcall target="usergenerateDD" /> <antcall target="generateDD" /> <antcall target="war" /> <antcall target="deploy" /> </target> </project>

​

**init**

<target name="init"> <mkdir dir="${build.dir}" /> <mkdir dir="${build.webinf.dir}" /> <mkdir dir="${build.webinf.dir}/tlds" /> <mkdir dir="${build.lib.dir}" /> <mkdir dir="${build.class.dir}" /> <mkdir dir="${build.src.dir}" /> <mkdir dir="${dist.dir}" /> <path id="xdoclet.class.path"> <fileset dir="${lib.xdoclet.dir}" includes="**/*.jar" /> <fileset dir="${lib.j2ee.dir}" includes="**/*.jar" /> <fileset dir="${lib.commons.dir}" includes="**/*.jar" /> <fileset dir="${lib.log4j.dir}" includes="**/*.jar" /> <fileset dir="${lib.jdbc.dir}" includes="**/*.jar" /> <fileset dir="${lib.dbpool.dir}" includes="**/*.jar" /> </path> <pathelement path="${build.class.dir}"/> <fileset dir="${lib.jdbc.dir}" includes="**/*.jar" /> <fileset dir="${lib.dbpool.dir}" includes="**/*.jar" /> </path> </target>

​

**copy**

<target name="copy" depends="init"> <copy todir="${build.webinf.dir}/lib"> <fileset dir="${lib.commons.dir}"/> <fileset dir="${lib.log4j.dir}"/> <fileset dir="${lib.jdbc.dir}"/> <fileset dir="${lib.dbpool.dir}"/> </copy> <copy todir="${build.class.dir}"> <fileset dir="${src.dir}"> <include name="**/*.properties"/> </fileset> </copy> <copy todir="${build.src.dir}"> <fileset dir="${src.dir}"> <include name="**/*.java"/> <include name="**/*.properties"/> </fileset> </copy> <copy todir="${build.dir}"> <fileset dir="${web.dir}"> <include name="**/*.jsp"/> </fileset> </copy> </target>

​

**war**

<target name="war" depends="init"> <war destFile="${dist.dir}/${war.name}" webxml="${build.dir}/WEB-INF/web.xml"> <fileset dir="${build.dir}" includes="**/*" /> </war> </target>

​

**clean**

<target name="clean"> <delete dir="${build.dir}" /> <delete dir="${dist.dir}" /> </target>

​

**deploy**

<target name="deploy"> <delete file="${env.CATALINA_HOME}/webapps/${war.name}" /> <delete dir="${env.CATALINA_HOME}/webapps/${deploy.dir}" /> <copy file="${dist.dir}/${war.name}" todir="${env.CATALINA_HOME}/webapps/"/> </target>

​

**resources**

<target name="resources"> <native2ascii src="${properties.dir}" dest="${resources.src.dir}" includes="**/*.properties"/> </target>

​

**dbinit**

<target name="dbinit" depends="init"> <sql driver="${drivers}" url="${oracle.url}" userid="${oracle.user}" password="${oracle.password}" src="${resource.dir}/createTable.sql" > </sql> </target>

​

**dbcopy**

<target name="dbcopy" depends="dbinit"> <copy todir="${build.src.dir}"> <fileset dir="${src.dir}"> <include name="**/*.java"/> </fileset> </copy> <copy todir="${build.class.dir}"> <fileset dir="${src.dir}"> <include name="**/dbpool.properties"/> </fileset> </copy> </target>

**complie**

<target name="compile" depends="init"> <javac srcdir="${build.src.dir}" destdir="${build.class.dir}" debug="true" deprecation="true"> <classpath refid="xdoclet.class.path"/> </javac> </target>

​

**usergenerateDD**

<target name="usergenerateDD" depends="init"> <delete file="${web.webinf.dir}/struts-config.xml"/> <delete file="${web.webinf.dir}/validation.xml"/> <delete file="${web.webinf.dir}/web.xml"/> <taskdef name="webdoclet" classname="xdoclet.modules.web.WebDocletTask" classpathref="xdoclet.class.path" /> <webdoclet destdir="${web.dir}" mergedir="${merge.dir}/user"> <fileset dir="${src.dir}"> <include name="**/user/**/*" /> </fileset> </webdoclet> </target>

​

**generateDD**

<target name="generateDD" depends="init"> <delete file="${web.webinf.dir}/struts-config.xml"/> <delete file="${web.webinf.dir}/web.xml"/> <taskdef name="webdoclet" classname="xdoclet.modules.web.WebDocletTask" classpathref="xdoclet.class.path" /> <webdoclet destdir="${web.dir}" mergedir="${merge.dir}/default"> <fileset dir="${src.dir}"> <include name="**/*.java" /> <exclude name="**/user/**/*" /> </fileset> <deploymentdescriptor servletspec="2.3" destdir="${build.webinf.dir}"> </deploymentdescriptor> </webdoclet> </target>
