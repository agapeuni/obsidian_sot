**ir_4_sender.ino**

#include <IRremote.h> int RECV_PIN = 3; IRrecv irrecv(RECV_PIN); decode_results results; int pins[] = { 4, 5, 6, 7, 8, 9 }; int pinLen = 6; void setup() { Serial.begin(9600); irrecv.enableIRIn(); for (int i = 0; i < pinLen; i++) { pinMode(pins[i], OUTPUT); } } void loop() { int idx = -1; if (irrecv.decode(&results)) { switch (results.value) { case 0xFF00FF: Serial.print("1"); idx = 0; break; case 0xFF807F: Serial.print("2"); idx = 1; break; case 0xFF40BF: Serial.print("3"); idx = 2; break; case 0xFFC03F: Serial.print("4"); idx = 3; break; case 0xFF20DF: Serial.print("5"); idx = 4; break; case 0xFFA05F: Serial.print("6"); idx = 5; break; } ledOn(pins[idx]); irrecv.resume(); } delay(100); } void ledOn(int pin) { digitalWrite(pin, HIGH); delay(500); digitalWrite(pin, LOW); }
