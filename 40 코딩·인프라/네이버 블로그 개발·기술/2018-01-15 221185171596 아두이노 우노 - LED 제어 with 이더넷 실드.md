**▣ 소스코드 : ethernet_4_pinstate.ino**

#include <Ethernet.h> #include <SPI.h> byte MAC_ADRS[] = { 0xB0, 0x12, 0xFA, 0x34, 0x1A, 0xE2 }; byte IP_ADRS[] = { 192, 168, 0, 15 }; EthernetServer ethernetServer(80); int numPins = 4; int pins[] = { 2, 3, 4, 5 }; 아래는 각각의 LED에 대한 On/Off 버튼으로 제어하는 코드입니다.

**▣ 소스코드 : ethernet_5_button.ino**

#include <Ethernet.h> #include <SPI.h> byte MAC_ADRS[] = { 0xB0, 0x12, 0xFA, 0x34, 0x1A, 0xE2 }; byte IP_ADRS[] = { 192, 168, 0, 15 }; byte GATEWAY[] = { 192, 168, 0, 1 }; byte SUBNET[] = { 255, 255, 255, 0 }; EthernetServer ethernetServer(80); int redled = 2; int yellowled = 3; int greenled = 4; int whiteled = 5; String buffer; void setup() { Serial.begin(9600); pinMode(redled, OUTPUT); pinMode(yellowled, OUTPUT); pinMode(greenled, OUTPUT); pinMode(whiteled, OUTPUT); Ethernet.begin(MAC_ADRS, IP_ADRS, GATEWAY, SUBNET); ethernetServer.begin(); Serial.print("server is at "); Serial.println(Ethernet.localIP()); } void loop() { EthernetClient client = ethernetServer.available(); if (client) { while (client.connected()) { if (client.available()) { char c = client.read(); if (buffer.length() < 100) { buffer += c; } if (c == '\n') { Serial.println(buffer); client.println("HTTP/1.1 200 OK"); client.println("Content-Type: text/html"); client.println(""); client.println("<h2>Control LEDs</h2>"); client.println(""); client.println("<h4 style='color: #f44336;'>Red LED</h4>"); client.println("<a href='/?r_on'><button style='background-color: #f44336; border-radius: 12px;'> ON</button></a>"); client.println("<a href='/?r_off'><button style='background-color: #f44336; border-radius: 12px;'> OFF</button></a>"); client.println(""); client.println("<h4 style='color: yellow;'>Yellow LED</h4>"); client.println("<a href='/?y_on'><button style='background-color: yellow; border-radius: 12px;'> ON</button></a>"); client.println("<a href='/?y_off'><button style='background-color: yellow; border-radius: 12px;'> OFF</button></a>"); client.println(""); client.println("<h4 style='color: lightgreen;'>Green LED</h4>"); client.println("<a href='/?g_on'><button style='background-color: lightgreen; border-radius: 12px;'> ON</button></a>"); client.println("<a href='/?g_off'><button style='background-color: lightgreen; border-radius: 12px;'> OFF</button></a>"); client.println(""); client.println("<h4>White LED</h4>"); client.println("<a href='/?w_on'><button style='border-radius: 12px;'> ON</button></a>"); client.println("<a href='/?w_off'><button style='border-radius: 12px;'> OFF</button></a>"); client.println(""); delay(10); client.stop(); if (buffer.indexOf("?r_on") > 0) { digitalWrite(redled, HIGH); } if (buffer.indexOf("?r_off") > 0) { digitalWrite(redled, LOW); } if (buffer.indexOf("?y_on") > 0) { digitalWrite(yellowled, HIGH); } if (buffer.indexOf("?y_off") > 0) { digitalWrite(yellowled, LOW); } if (buffer.indexOf("?g_on") > 0) { digitalWrite(greenled, HIGH); } if (buffer.indexOf("?g_off") > 0) { digitalWrite(greenled, LOW); } if (buffer.indexOf("?w_on") > 0) { digitalWrite(whiteled, HIGH); } if (buffer.indexOf("?w_off") > 0) { digitalWrite(whiteled, LOW); } buffer = ""; } } } } }
