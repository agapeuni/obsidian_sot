**1\. 개요**

Node.js에서 제공하는 url 모듈을 사용하여 URL 문자열을 파싱하고, 그 결과를 객체 형태로 분석하는 방법을 소개하고자 한다. URL은 웹 자원에 대한 경로를 지정하는 구조화된 문자열로, 다양한 구성 요소를 포함하고 있다. 이러한 요소들을 객체로 변환하면 URL에 포함된 정보를 보다 효과적으로 추출하고 활용할 수 있다.

​

​

**2\. 사용법**

URL 문자열은 여러 구성 요소를 포함하는 구조화된 문자열이다. URL 문자열은 일반적으로 프로토콜, 호스트, 포트, 경로, 쿼리 스트링, 해시 등의 요소로 구성된다. Node.js에서는 require('url') 구문을 통해 url 모듈을 불러오며, 이 모듈의 parse() 메서드를 사용하면 URL을 분석할 수 있다.

![](https://postfiles.pstatic.net/MjAyNTA0MTlfMTc1/MDAxNzQ1MDY3Nzg3NjQy.zODBMr19vFhruW9OXymduua6AOex8_uzx8FBQuSQs5gg.EvLnP_eNcO2rH9gQeIf4LGetkwHBlddUFIlWnTrzL4og.PNG/img1.daumcdn.png?type=w80_blur)

​

​

url 모듈을 선언하고 parse() 함수로 주소를 파싱한다. parse() 함수는 첫 번째 인자로 URL 문자열을, 두 번째 인자로 true를 지정하면 쿼리 문자열을 객체로 파싱해준다. 세 번째 인자는 슬래시 이후의 경로를 해석하는 방식을 설정한다.

var url = require('url'); var urlObj = url.parse("http://domain:8080/admin/page?name=admin&pwd=1234#top", true, true);

​

파싱된 객체를 사용하여 아래와 같이 각각의 정보를 표시한다.

console.log("urlObj.href = " + urlObj.href); // 전체 URL console.log("urlObj.protocol = " + urlObj.protocol); // 프로토콜 console.log("urlObj.host = " + urlObj.host); // 호스트 console.log("urlObj.port = " + urlObj.port); // 포트 console.log("urlObj.path = " + urlObj.path); // 전체 경로 console.log("urlObj.pathname = " + urlObj.pathname); // 경로 console.log("urlObj.search = " + urlObj.search); // 검색 문자열 console.log("urlObj.query :"); console.log(urlObj.query); // 쿼리 객체 console.log("urlObj.hash = " + urlObj.hash); // 해시 값

​

**3\. 예제**

위에서 설명한 코드로 실행한 예제이다.

/** url : Uniform Resource Locator */ var url = require('url'); var urlObj = url.parse("http://domain8080/admin/page?name=admin&pwd=1234#top", true, true); console.log("urlObj.href = " + urlObj.href); console.log("urlObj.protocol = " + urlObj.protocol); console.log("urlObj.host = " + urlObj.host); console.log("urlObj.port = " + urlObj.port); console.log("urlObj.path = " + urlObj.path); console.log("urlObj.pathname = " + urlObj.pathname); console.log("urlObj.search = " + urlObj.search); console.log("urlObj.query :"); console.log(urlObj.query); console.log("urlObj.hash = " + urlObj.hash); console.log(url.format({ protocol : 'http', host : 'domain', pathname : '/' }));

실행 결과

urlObj.href = http://domain8080/admin/page?name=admin&pwd=1234#top urlObj.protocol = http: urlObj.host = domain8080 urlObj.port = null urlObj.path = /admin/page?name=admin&pwd=1234 urlObj.pathname = /admin/page urlObj.search = ?name=admin&pwd=1234 urlObj.query : [Object: null prototype] { name: 'admin', pwd: '1234' } urlObj.hash = #top http://domain/
