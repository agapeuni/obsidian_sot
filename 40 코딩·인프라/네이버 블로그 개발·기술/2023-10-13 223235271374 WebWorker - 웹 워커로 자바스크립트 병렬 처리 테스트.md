웹워커를 테스트하기 위해서 텍스트 박스와 라벨을 표시하는 <span> 태그 3개를 만들어 두었다. 

텍스트 박스에서 값을 입력하면 각각의 웹워커에서 작업을 하고 나서 결과값을 전송해 <span>에 표시한다.

* * *

파일 : WebWorker.html 

<!DOCTYPE html> <html> <head> <meta charset="utf-8"> <title>WebWorker</title> <style type="text/css" media="screen"> span { word-wrap: break-word; } </style> <script type="text/javascript"> var helloWorker1; var helloWorker2; var htmlLabel; var workerLabel1; var workerLabel2; window.onload = initWorker; function initWorker() { htmlLabel = document.getElementById("htmlHello"); htmlLabel.innerHTML = "HTML : "; workerLabel1 = document.getElementById("workerHello1"); workerLabel2 = document.getElementById("workerHello2"); workerLabel3 = document.getElementById("workerHello3"); workerLabel1.innerHTML = "Worker1 : "; workerLabel2.innerHTML = "Worker2 : "; workerLabel3.innerHTML = "Worker2 : "; helloWorker1 = new Worker("WebWorker1.js"); helloWorker2 = new Worker("WebWorker2.js"); helloWorker3 = new Worker("WebWorker3.js"); helloWorker1.onmessage = function(event) { workerLabel1.innerHTML = event.data; } helloWorker2.onmessage = function(event) { workerLabel2.innerHTML = event.data; } helloWorker3.onmessage = function(event) { workerLabel3.innerHTML = event.data; } } function callSayHello(objInput) { htmlLabel.innerHTML = "HTML : " + objInput.value; // Worker로 값을 전송. helloWorker1.postMessage(objInput.value); helloWorker2.postMessage(objInput.value); helloWorker3.postMessage(objInput.value); } </script> </head> <body> 값을 입력해 주세요. <br /> <input type="text"onkeyup="callSayHello(this);"> <br /> <br /> <span id="htmlHello"></span> <br /> <span id="workerHello1"></span> <br /> <span id="workerHello2"></span> <br /> <span id="workerHello3"></span> <br /> </body> </html>

**웹워커1 (WebWorker1.js)**

화면에서 메시지를 받고 나서 3개의 for 문으로 가상의 작업을 가정하여 임의로 만들어 두었다. 

충분한 시간의 작업을 하고 나서 화면으로 retval 결과값을 전송한다.

// 화면에서 메세지를 받음 self.onmessage = function(event) { var retval = "Worker1 : " + event.data; var len = 1000; var z = 0; for (var i = 0; i < len; i++) { for (var j = 0; j < len; j++) { for (var k = 0; k < len; k++) { z = i + j * 2 - k; } } } // 화면으로 메세지를 전송. self.postMessage(retval); };
