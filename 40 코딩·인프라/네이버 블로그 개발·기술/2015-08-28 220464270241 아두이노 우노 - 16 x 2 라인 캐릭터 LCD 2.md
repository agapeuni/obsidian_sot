**▣ 소스코드 : lcd_char_3_progress_left.ino**

#include <LiquidCrystal.h> LiquidCrystal lcd(2, 3, 4, 5, 6, 7); byte b00[8] = { 00, 00, 00, 00, 00, 00, 00, 00 }; byte b16[8] = { 16, 16, 16, 16, 16, 16, 16, 16 }; byte b24[8] = { 24, 24, 24, 24, 24, 24, 24, 24 }; byte b28[8] = { 28, 28, 28, 28, 28, 28, 28, 28 }; byte b30[8] = { 30, 30, 30, 30, 30, 30, 30, 30 }; byte b31[8] = { 31, 31, 31, 31, 31, 31, 31, 31 }; int delayTime = 15; void setup() { lcd.begin(16, 2); lcd.createChar(0, b00); lcd.createChar(1, b16); lcd.createChar(2, b24); lcd.createChar(3, b28); lcd.createChar(4, b30); lcd.createChar(5, b31); } void loop() { // 0 Line --> for (int i = 0; i < 16; i++) { for (int j = 0; j < 6; j++) { lcd.setCursor(i, 0); lcd.write(j); delay(delayTime); } } // 1 Line --> for (int i = 0; i < 16; i++) { for (int j = 0; j < 6; j++) { lcd.setCursor(i, 1); lcd.write(j); delay(delayTime); } } // 1 Line <\-- for (int i = 15; i > -1; i--) { for (int j = 5; j > -1; j--) { lcd.setCursor(i, 1); lcd.write(j); delay(delayTime); } } // 0 Line <\-- for (int i = 15; i > -1; i--) { for (int j = 5; j > -1; j--) { lcd.setCursor(i, 0); lcd.write(j); delay(delayTime); } } }

진행바가 윗줄 왼쪽에서 시작하여 점점 증가하여 아래줄까지 진행하고 나서 다시 이전으로 돌아가는 것을 볼 수 있습니다.

**▣ 소스코드 : lcd_char_4_progress_right.ino**

#include <LiquidCrystal.h> LiquidCrystal lcd(2, 3, 4, 5, 6, 7); byte b00[8] = { 00, 00, 00, 00, 00, 00, 00, 00 }; byte b01[8] = { 01, 01, 01, 01, 01, 01, 01, 01 }; byte b03[8] = { 03, 03, 03, 03, 03, 03, 03, 03 }; byte b07[8] = { 07, 07, 07, 07, 07, 07, 07, 07 }; byte b15[8] = { 15, 15, 15, 15, 15, 15, 15, 15 }; byte b31[8] = { 31, 31, 31, 31, 31, 31, 31, 31 }; int delayTime = 15; void setup() { lcd.begin(16, 2); lcd.createChar(0, b00); lcd.createChar(1, b01); lcd.createChar(2, b03); lcd.createChar(3, b07); lcd.createChar(4, b15); lcd.createChar(5, b31); } void loop() { // 0 Line <\-- for (int i = 15; i > -1; i--) { for (int j = 0; j < 6; j++) { lcd.setCursor(i, 1); lcd.setCursor(i, 0); lcd.write(j); delay(delayTime); } } // 1 Line <\-- for (int i = 15; i > -1; i--) { for (int j = 0; j < 6; j++) { lcd.setCursor(i, 1); lcd.setCursor(i, 1); lcd.write(j); delay(delayTime); } } // 1 Line --> for (int i = 0; i < 16; i++) { for (int j = 5; j > -1; j--) { lcd.setCursor(i, 1); lcd.write(j); delay(delayTime); } } // 0 Line --> for (int i = 0; i < 16; i++) { for (int j = 5; j > -1; j--) { lcd.setCursor(i, 0); lcd.write(j); delay(delayTime); } } }

이번에는 진행바가 윗줄 오른쪽에서 시작하여 왼쪽으로 점점 증가하는 동영상입니다.
