1\. 체크박스 체크 확인

$('input[type="checkbox"]'); // 모든 체크박스 선택 $('#checkboxID'); // 특정 ID를 가진 체크박스 선택 $('.checkboxClass'); // 특정 클래스를 가진 체크박스 선택

​

2\. 체크박스 선택 상태 확인 및 설정

내용을 입력하세요.if ($('#checkboxID').is(':checked')) { // 체크박스가 체크된 상태일 때 실행 } var ischecked = $('#mod_gb').attr('checked'); if(ischecked) { alert("체크되었습니다."); } else { alert("체크되지 않았습니다."); }

$('#checkboxID').prop('checked', true); // 체크박스를 체크 상태로 설정 $('#checkboxID').prop('checked', false); // 체크박스를 체크 해제 상태로 설정 

​

3\. 체크박스 전체 선택

// name이 mod_gb인 체크박스 그룹의 아무 박스를 선택하면 $('input[name=mod_gb]:checkbox').click(function() { var gb1 = $('#mod_gb1').attr('checked'); var gb2 = $('#mod_gb2').attr('checked'); var gb3 = $('#mod_gb3').attr('checked'); if(gb3 && (gb1 || gb2)){ alert("ID 해지는 다른 항목과 중복 선택하실 수 없습니다."); $('#mod_gb3').attr('checked', false); } });

$('input[type="checkbox"]').prop('checked', true); // 모든 체크박스를 선택 $('input[type="checkbox"]').prop('checked', false); // 모든 체크박스를 해제

​

4\. 체크박스 전체 선택 및 해제 기능

// 특정 체크박스 클릭 시 전체 체크박스 선택/해제하기 $('#selectAll').on('click', function() { $('input[type="checkbox"]').prop('checked', $(this).is(':checked')); });

​

5\. 체크박스 변경 이벤트 처리

// 체크박스 상태가 변경될 때마다 이벤트 처리 $('input[type="checkbox"]').on('change', function() { if ($(this).is(':checked')) { console.log('체크됨'); } else { console.log('체크 해제됨'); } });

​

6\. 체크박스 선택 확인

$('#btn').click(function(){ var cnt = $("input[name=mod_gb]:checkbox:checked").length; if(cnt < 1){ alert('한 개 이상을 선택하셔야 합니다'); }else{ alert(cnt+'개가 선택되었습니다'); } });

​

7\. 체크박스 선택된 값 확인하기

var selectedValues = []; $('input[type="checkbox"]:checked').each(function() { selectedValues.push($(this).val()); }); console.log('선택된 값:', selectedValues);
