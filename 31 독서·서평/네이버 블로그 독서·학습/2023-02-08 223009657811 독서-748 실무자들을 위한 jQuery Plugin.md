실무자를 위한다고 해서 나름 기대하고 보았는데 중후반까지 JavaScript와 jQuery 기초에 대한 초보적인 내용이 주를 이룬다. 이후 100여 페이지가 jQuery 플러그인에 대한 내용인데 제작에 대한 것보다 간단한 플러그인의 개요와 사용법이 전부이다. 책의 제목에는 "실무자들을 위한~"이라고 되어 있는데 실무자를 위한 것이 아니라 입문자가 대상으로 여겨진다. 저자에게는 미안하지만 jQuery Plugin이라는 제목이 아깝다는 생각이 든다.

​

​

간단한 jQuery 플러그인을 하나의 파일로 만들어 보았다. 

아래의 형태로 사용해서 플러그인을 쉽게 만들 수가 있다.

<html> <head> <title>jQuery Plugin</title> <script type="text/javascript" src="http://code.jquery.com/jquery-latest.js"></script> <script> // 자체실행 익명함수(Immediately Invoked Function Expression) (function($) { // 플러그인 이름선언 $.fn.shakeMenu = function(options) { // 기본값을 설정 var defaults = { shakeLeft : 100, defaultLeft : 10, evenColor : '#ddd', oddColor : '#eee', }; // 옵션값이 있으면 사용 var settings = $.extend( {}, defaults, options ); // 엘리먼트 반복 (메서드 체이닝) return this.each(function() { // 코드 작성부분 var s = settings; var obj = $(this); var items = $("li", obj); $("li:odd", obj).css('background-color', s.oddColor); $("li:even", obj).css('background-color', s.evenColor); items.mouseover(function() { $(this).animate({ paddingLeft : s.shakeLeft }, 500); }).mouseout(function() { $(this).animate({ paddingLeft : s.defaultLeft }, 500); }); items.click(function() { $("#view").text($(this).text()); }) }); }; })(jQuery); </script> <script type="text/javascript"> $(document).ready(function() { $('#menu').shakeMenu({ shakeLeft : 300, defaultLeft : 3 }); }); </script> <style> a { text-decoration: none } #menu { list-style: none; width: 400px; } #menu li { margin: 3px; padding: 3px; } </style> </head> <body> <br/><br/> <ul id="menu"> <li>Menu 1</li> <li>Menu 2</li> <li>Menu 3</li> <li>Menu 4</li> <li>Menu 5</li> <li>Menu 6</li> <li>Menu 7</li> </ul> <span id="view"></span> </body> </html>
